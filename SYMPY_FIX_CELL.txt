import subprocess
import sys

print("Fixing SymPy compatibility issue...")

# Uninstall old SymPy
subprocess.run([sys.executable, "-m", "pip", "uninstall", "-y", "sympy"],
              capture_output=True)

# Install compatible version
result = subprocess.run([sys.executable, "-m", "pip", "install", "-q", "sympy>=1.12"],
                       capture_output=True, text=True)

if result.returncode == 0:
    print("✓ SymPy upgraded successfully")
    print("")
    print("="*80)
    print("NEXT STEP: Runtime > Restart runtime")
    print("Then re-run all cells from the beginning")
    print("="*80)
else:
    print("✗ Error:", result.stderr)
